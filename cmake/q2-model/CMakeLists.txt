# Experimental first-model-test executable only. NO server or production target.
cmake_minimum_required(VERSION 3.21)
project(lie_q2_model_test LANGUAGES C CXX HIP)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_HIP_STANDARD 20)
set(CMAKE_HIP_STANDARD_REQUIRED ON)
get_filename_component(LIE_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
if(NOT EXISTS "${GUFO_SOURCE}/LIE-Q2-MODEL-TEST-SOURCE.json")
  message(FATAL_ERROR "Use the isolated first-model-test source preparer")
endif()
if(NOT LIE_BUILD_ID MATCHES "^[a-z0-9-]+$")
  message(FATAL_ERROR "Test build identity is required")
endif()
add_compile_definitions(_POSIX_C_SOURCE=200809L LIE_BUILD_ID="${LIE_BUILD_ID}")
set(ENGINE_ENABLE_HIP ON CACHE BOOL "" FORCE)
add_subdirectory(../gufo-runtime provider)
target_include_directories(gufo_qwen38_flash_next PRIVATE "${LIE_ROOT}/tests")
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-c)
add_library(q2-test-memory STATIC "${LIE_ROOT}/tests/q2_model_memory.c")
target_include_directories(q2-test-memory PUBLIC "${LIE_ROOT}/tests")
target_compile_options(q2-test-memory PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(q2-test-guard STATIC "${LIE_ROOT}/tests/q2_model_guard.cpp")
target_compile_options(q2-test-guard PRIVATE -Wall -Wextra -Werror)
target_link_libraries(q2-test-guard PUBLIC q2-test-memory gufo_qwen38_flash_next)
target_link_options(q2-test-guard INTERFACE -Wl,--wrap=hipMalloc -Wl,--wrap=hipHostMalloc)
add_library(q2-test-adapter STATIC "${LIE_ROOT}/adapters/gufo.cpp"
  "${LIE_ROOT}/adapters/gufo_binding.c" "${LIE_ROOT}/adapters/gufo_device.cpp")
target_compile_definitions(q2-test-adapter PRIVATE LIE_GUFO_ADAPTER_OPT_IN=1)
target_include_directories(q2-test-adapter PUBLIC "${LIE_ROOT}/include")
target_link_libraries(q2-test-adapter PUBLIC q2-test-guard gufo_qwen38_flash_next)
add_executable(q2-model-first-test "${LIE_ROOT}/tests/q2_model_probe.c")
target_compile_options(q2-model-first-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(q2-model-first-test PRIVATE q2-test-adapter PkgConfig::JSON m)
add_executable(q2-model-header-plan "${LIE_ROOT}/tests/q2_header_bind.cpp")
target_compile_definitions(q2-model-header-plan PRIVATE LIE_Q2_MODEL_TEST_PLAN=1)
target_compile_options(q2-model-header-plan PRIVATE -UNDEBUG -Wall -Wextra -Werror)
target_link_libraries(q2-model-header-plan PRIVATE q2-test-guard)
add_executable(test-q2-model-memory "${LIE_ROOT}/tests/test_q2_model_memory.c")
target_compile_options(test-q2-model-memory PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(test-q2-model-memory PRIVATE q2-test-memory)
