cmake_minimum_required(VERSION 3.21)
project(lie_core_model_flow LANGUAGES C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(READ "${ROOT}/config/ds4-walk-bench-source.json" parent)
file(READ "${ROOT}/config/core-model-flow-source.json" source)
file(SHA256 "${ROOT}/config/ds4-walk-bench-source.json" parent_actual)
string(JSON parent_expected GET "${source}" parent_manifest_sha256)
if(NOT parent_actual STREQUAL parent_expected)
  message(FATAL_ERROR "Parent source manifest changed")
endif()
file(SHA256 "${ROOT}/experiments/core-model-flow.patch" patch_actual)
string(JSON patch_expected GET "${source}" patch_sha256)
if(NOT patch_actual STREQUAL patch_expected)
  message(FATAL_ERROR "Model-flow patch changed")
endif()
string(JSON changed_count LENGTH "${source}" changed_files)
math(EXPR changed_last "${changed_count}-1")
set(changed_names)
foreach(i RANGE 0 ${changed_last})
  string(JSON name MEMBER "${source}" changed_files ${i})
  list(APPEND changed_names "${name}")
endforeach()
string(JSON count LENGTH "${parent}" files)
math(EXPR last "${count}-1")
foreach(i RANGE 0 ${last})
  string(JSON name MEMBER "${parent}" files ${i})
  if(name IN_LIST changed_names)
    string(JSON expected GET "${source}" changed_files "${name}")
  else()
    string(JSON expected GET "${parent}" files "${name}")
  endif()
  file(SHA256 "${ROOT}/.deps/lie-core-model-flow/${name}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Model-flow source changed: ${name}")
  endif()
endforeach()
file(READ "${ROOT}/config/q2-counting-bench-build.json" archives)
string(JSON count LENGTH "${archives}" retained_archives)
math(EXPR last "${count}-1")
foreach(i RANGE 0 ${last})
  string(JSON name MEMBER "${archives}" retained_archives ${i})
  string(JSON expected GET "${archives}" retained_archives "${name}")
  file(SHA256 "${ROOT}/${name}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Retained numerical archive changed: ${name}")
  endif()
endforeach()
set(LIE_GUFO_RUNTIME OFF CACHE BOOL "" FORCE)
set(LIE_C17_SAMPLING OFF CACHE BOOL "" FORCE)
set(LIE_MTP OFF CACHE BOOL "" FORCE)
set(LIE_VISION OFF CACHE BOOL "" FORCE)
set(LIE_DS4_RUNTIME_CACHE OFF CACHE BOOL "" FORCE)
set(LIE_KVC_INTERCHANGE OFF CACHE BOOL "" FORCE)
set(LIE_SYSTEM_LIBUV ON CACHE BOOL "" FORCE)
set(LIE_BUILD_ID core-model-flow CACHE STRING "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/.deps/lie-core-model-flow" core)

list(FILTER bench_sources EXCLUDE REGEX "executor_unavailable\\.c$")
set_property(TARGET synapse-lie-bench PROPERTY SOURCES "${bench_sources}")
set_property(TARGET synapse-lie-bench PROPERTY LINKER_LANGUAGE CXX)
set(RETAINED "${ROOT}/.deps/build-prefill-chunks")
target_link_libraries(synapse-lie-bench PRIVATE
  "${RETAINED}/cmake/curve/libq2_curve_adapter.a"
  lie_qwen_state lie_store

  "${RETAINED}/libq2_host.a"

  "${ROOT}/.deps/build-hc-scalar-matched/cmake/hip/qwen/libgufo_qwen38_flash_next_mmq.a"
  /opt/rocm/lib/libhipblas.so /opt/rocm/lib/librocblas.so
  /opt/rocm/lib/libamdhip64.so /opt/rocm/lib/libhipblaslt.so)
