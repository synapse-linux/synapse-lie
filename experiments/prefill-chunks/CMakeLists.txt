cmake_minimum_required(VERSION 3.21)
project(q2_prefill_chunk_contracts LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(LIE_CORE_ONLY ON CACHE BOOL "" FORCE)
set(LIE_MTP OFF CACHE BOOL "" FORCE)
set(LIE_VISION OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING ON CACHE BOOL "" FORCE)
set(LIE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(CHUNK_CORE "${LIE_ROOT}/.deps/lie-q2-prefill-chunks")
set(CHUNK_GPU "${LIE_ROOT}/.deps/gufo-q2-prefill-chunks/src/models/qwen38_flash_next/kernels/rocm")
enable_testing()
add_subdirectory("${CHUNK_CORE}" core)
if(LIE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_definitions(_POSIX_C_SOURCE=200809L LIE_MTP=0 LIE_VISION=0)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_executable(test-prefill-chunk-shapes "${LIE_ROOT}/tests/q2_prefill_chunk_shapes.c"
  "${CHUNK_GPU}/iq2_mixed_tiles.c")
target_include_directories(test-prefill-chunk-shapes PRIVATE "${CHUNK_GPU}")
add_test(NAME prefill-chunk-shapes COMMAND test-prefill-chunk-shapes)
add_executable(test-prefill-chunk-core "${LIE_ROOT}/tests/q2_prefill_chunk_core.c"
  "${CHUNK_CORE}/tests/fake_executor.c")
target_include_directories(test-prefill-chunk-core PRIVATE "${CHUNK_CORE}/tests")
target_link_libraries(test-prefill-chunk-core PRIVATE lie_core)
add_test(NAME prefill-chunk-core COMMAND test-prefill-chunk-core)
set_tests_properties(prefill-chunk-core prefill-chunk-shapes PROPERTIES TIMEOUT 120)
