cmake_minimum_required(VERSION 3.21)
project(synapse_lie VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(LIE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(LIE_GUFO_HEADER_CHECK "Compile opt-in transitional Gufo adapter (object only; no link/inference)" OFF)
option(LIE_GUFO_RUNTIME "Link explicitly selected embedded Gufo HIP runtime" OFF)
set(GUFO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.deps/gufo-f783fedb" CACHE PATH "Pinned, project-owned upstream source")
set(GUFO_BUILD "" CACHE PATH "Verified private build from tools/build-gufo.py")
find_package(PkgConfig REQUIRED)
pkg_check_modules(UV REQUIRED IMPORTED_TARGET libuv)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-c)
pkg_check_modules(HTTP REQUIRED IMPORTED_TARGET libllhttp)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
find_package(Threads REQUIRED)
set(LIE_BUILD_ID "unrecorded" CACHE STRING "Evidence/build label, not hardware qualification")
if(NOT LIE_BUILD_ID MATCHES "^[A-Za-z0-9_.-]+$")
  message(FATAL_ERROR "LIE_BUILD_ID must be an alphanumeric evidence label")
endif()
add_compile_definitions(_POSIX_C_SOURCE=200809L LIE_BUILD_ID="${LIE_BUILD_ID}")
add_compile_options(-Wall -Wextra -Wpedantic -Werror -fno-common)
if(LIE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(lie_flow STATIC src/flow.c)
target_include_directories(lie_flow PUBLIC include)
target_link_libraries(lie_flow PUBLIC Threads::Threads)
add_library(lie_metrics STATIC src/metrics.c)
target_include_directories(lie_metrics PUBLIC include)
target_link_libraries(lie_metrics PUBLIC PkgConfig::JSON Threads::Threads m)
add_library(lie_runtime STATIC src/chat.c src/worker.c src/wire.c)
target_include_directories(lie_runtime PUBLIC include)
target_link_libraries(lie_runtime PUBLIC lie_flow PkgConfig::JSON Threads::Threads)
add_executable(synapse-lie-server src/server.c)
target_link_libraries(synapse-lie-server PRIVATE lie_metrics lie_runtime PkgConfig::UV PkgConfig::HTTP)
if(LIE_GUFO_RUNTIME)
  enable_language(CXX)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tools/check-gufo-build.py
    "${GUFO_SOURCE}" "${GUFO_BUILD}" RESULT_VARIABLE verified)
  if(NOT verified EQUAL 0)
    message(FATAL_ERROR "Gufo source/build verification failed")
  endif()
  list(APPEND CMAKE_PREFIX_PATH /opt/rocm)
  find_package(hip REQUIRED)
  find_package(hipblas REQUIRED)
  find_package(hipblaslt REQUIRED)
  find_package(rocblas REQUIRED)
  find_package(ICU REQUIRED COMPONENTS uc i18n)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  find_package(PNG REQUIRED)
  find_package(JPEG REQUIRED)
  add_library(lie_gufo STATIC adapters/gufo.cpp adapters/gufo_binding.c adapters/gufo_device.cpp)
  target_compile_definitions(lie_gufo PRIVATE LIE_GUFO_ADAPTER_OPT_IN=1)
  target_compile_features(lie_gufo PRIVATE cxx_std_20)
  target_include_directories(lie_gufo PRIVATE include "${GUFO_SOURCE}")
  target_link_libraries(lie_gufo PUBLIC
    "-Wl,--start-group"
    "${GUFO_BUILD}/src/models/qwen38_flash_next/libgufo_qwen38_flash_next.a"
    "${GUFO_BUILD}/src/models/qwen38_flash_next/libgufo_qwen38_flash_next_mmq.a"
    "${GUFO_BUILD}/libgufo_core.a" "-Wl,--end-group"
    hip::host roc::hipblas roc::hipblaslt roc::rocblas ICU::uc ICU::i18n
    OpenSSL::Crypto PNG::PNG JPEG::JPEG PkgConfig::CURL Threads::Threads)
  target_link_libraries(synapse-lie-server PRIVATE lie_gufo)
  add_executable(lie-executor-bench tools/executor-bench.c)
  target_include_directories(lie-executor-bench PRIVATE include)
  target_link_libraries(lie-executor-bench PRIVATE lie_gufo PkgConfig::JSON OpenSSL::Crypto m)
else()
  target_sources(synapse-lie-server PRIVATE src/executor_unavailable.c)
endif()
add_executable(synapse-lie-monitor src/monitor.c src/prometheus.c)
target_link_libraries(synapse-lie-monitor PRIVATE PkgConfig::CURL PkgConfig::JSON m)
target_include_directories(synapse-lie-monitor PRIVATE include)
