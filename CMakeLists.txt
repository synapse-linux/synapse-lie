function(lie_python_test name script)
  if(LIE_LEGACY_PYTHON_TESTS)
    add_test(NAME ${name} COMMAND ${Python3_EXECUTABLE} -B "${script}" ${ARGN})
    set_tests_properties(${name} PROPERTIES TIMEOUT 60)
  endif()
endfunction()
function(lie_enable_test_assertions)
  # Test assertions contain oracle checks and fixture setup. Keep them active
  # in optimized configurations without changing production library flags.
  if(BUILD_TESTING)
    get_property(lie_test_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(lie_test_target IN LISTS lie_test_targets)
      if(lie_test_target MATCHES "^test-")
        target_compile_options(${lie_test_target} PRIVATE -UNDEBUG)
      endif()
    endforeach()
  endif()
endfunction()
