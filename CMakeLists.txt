cmake_minimum_required(VERSION 3.21)
project(synapse_lie VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(LIE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(LIE_GUFO_HEADER_CHECK "Compile reference-only Gufo interop (not the LIE backend; no link/inference)" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(UV REQUIRED IMPORTED_TARGET libuv)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-c)
pkg_check_modules(HTTP REQUIRED IMPORTED_TARGET libllhttp)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
find_package(Threads REQUIRED)
add_compile_definitions(_POSIX_C_SOURCE=200809L)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -fno-common)
if(LIE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(lie_flow STATIC src/flow.c)
target_include_directories(lie_flow PUBLIC include)
target_link_libraries(lie_flow PUBLIC Threads::Threads)
add_library(lie_metrics STATIC src/metrics.c)
target_include_directories(lie_metrics PUBLIC include)
target_link_libraries(lie_metrics PUBLIC PkgConfig::JSON Threads::Threads m)
add_executable(synapse-lie-server src/server.c)
target_link_libraries(synapse-lie-server PRIVATE lie_metrics PkgConfig::UV PkgConfig::HTTP)
add_executable(synapse-lie-monitor src/monitor.c src/prometheus.c)
target_link_libraries(synapse-lie-monitor PRIVATE PkgConfig::CURL PkgConfig::JSON m)
target_include_directories(synapse-lie-monitor PRIVATE include)
include(CTest)
if(BUILD_TESTING)
  add_executable(test-flow tests/test_flow.c)
  target_link_libraries(test-flow PRIVATE lie_flow)
  add_test(NAME reactive-flow COMMAND test-flow)
  set_tests_properties(reactive-flow PROPERTIES TIMEOUT 20)
  add_executable(test-metrics tests/test_metrics.c)
  target_link_libraries(test-metrics PRIVATE lie_metrics)
  add_test(NAME metrics COMMAND test-metrics)
  add_executable(test-prometheus tests/test_prometheus.c src/prometheus.c)
  target_include_directories(test-prometheus PRIVATE include)
  target_link_libraries(test-prometheus PRIVATE m)
  add_test(NAME monitor-parser COMMAND test-prometheus)
  add_executable(test-executor-abi tests/test_executor_abi.c)
  target_include_directories(test-executor-abi PRIVATE include)
  add_test(NAME executor-c-layout COMMAND test-executor-abi)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME http-monitor COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_http.py $<TARGET_FILE:synapse-lie-server> $<TARGET_FILE:synapse-lie-monitor>)
  set_tests_properties(http-monitor PROPERTIES TIMEOUT 45)
endif()
if(LIE_GUFO_HEADER_CHECK)
  enable_language(CXX)
  set(GUFO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.deps/gufo-f783fedb" CACHE PATH "Pinned, project-owned upstream source")
  # Reference experiment only. Never link this target into the LIE server.
  add_library(lie_gufo_header_check OBJECT adapters/gufo.cpp)
  target_compile_definitions(lie_gufo_header_check PRIVATE LIE_REFERENCE_ADAPTER_ONLY=1)
  target_compile_features(lie_gufo_header_check PRIVATE cxx_std_20)
  target_include_directories(lie_gufo_header_check PRIVATE include "${GUFO_SOURCE}")
endif()
